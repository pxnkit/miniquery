cmake_minimum_required(VERSION 3.16)
project(miniquery VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
add_executable(miniquery src/main.cpp)
target_link_libraries(miniquery PRIVATE Threads::Threads)
if(MSVC)
  target_compile_options(miniquery PRIVATE /W4)
else()
  target_compile_options(miniquery PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME query_correctness COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_query.py $<TARGET_FILE:miniquery>)
